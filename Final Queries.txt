CREATE TABLE vehicles( 
vehicle_id SERIAL PRIMARY KEY NOT NULL, 
year SMALLINT NOT NULL, 
VIN VARCHAR(17) UNIQUE NOT NULL,
make VARCHAR(50) NOT NULL, 
model VARCHAR(100) NOT NULL 
); 
CREATE TABLE listings ( 
vehicle_id INTEGER REFERENCES vehicles(vehicle_id), 
price INTEGER NOT NULL, 
mileage INTEGER NOT NULL,
profit NUMERIC
); 
CREATE TABLE regions ( 
vehicle_id INTEGER REFERENCES vehicles(vehicle_id), 
city VARCHAR(100) NOT NULL, 
state VARCHAR(5) NOT NULL, 
region VARCHAR(20) 
);  

SELECT * FROM vehicles; --  65499 -- vehicle_id, year, vin, make, model
SELECT * FROM listings; --  65499 -- vehicle_id, price, mileage, profit
select * from regions; --  65499 -- vehicle_id, city, state, region

/* Combine all 3 tables to create a full vehicles list and include all necessary fields */

SELECT ve.vehicle_id, make, model, year, li.mileage, price, profit, re.city, state, region  
INTO vehicles_list
FROM vehicles ve
JOIN listings li ON li.vehicle_id = ve.vehicle_id
JOIN regions re ON re.vehicle_id = li.vehicle_id;

/* Which car models are oversupplied */
SELECT make, model, COUNT(*) AS vehicle_count
INTO oversupplied_cars_count
FROM vehicles_list
GROUP BY make, model
HAVING COUNT(*) > 2000
;
